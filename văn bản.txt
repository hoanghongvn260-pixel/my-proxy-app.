<script>
    // 1. TỰ ĐỘNG LẤY IP KHI MỞ TRANG
    function layIP() {
        fetch('https://api.ipify.org?format=json')
            .then(res => res.json())
            .then(data => {
                document.getElementById('ipDisplay').value = data.ip;
            })
            .catch(() => {
                document.getElementById('ipDisplay').value = "Không thể lấy IP";
            });
    }

    // Chạy hàm lấy IP ngay khi tải trang
    window.onload = layIP;

    // 2. XỬ LÝ KÍCH HOẠT KEY
    function kichHoat() {
        const key = document.getElementById('keyInput').value;
        const msg = document.getElementById('status-msg');

        if (key.trim() === "") {
            msg.innerHTML = "❌ <span style='color:red'>Vui lòng nhập Mã Key trước!</span>";
            return;
        }

        msg.innerHTML = "<div class='loading-spinner'></div> Đang xác thực Key trên hệ thống...";
        
        // Giả lập thời gian chờ xử lý 2.5 giây
        setTimeout(() => {
            if (key === "Phong12345") { // Đây là mã key "mẹo" để test
                msg.innerHTML = "✅ <span style='color:#28a745'>Kích hoạt thành công! Proxy đã được mở cho IP của bạn.</span>";
            } else {
                msg.innerHTML = "❌ <span style='color:red'>Lỗi: Mã Key '" + key + "' không tồn tại hoặc đã bị dùng.</span>";
            }
        }, 2500);
    }

    // 3. XỬ LÝ KIỂM TRA HỆ THỐNG
    function kiemTraHeThong() {
        const msg = document.getElementById('status-msg');
        msg.innerHTML = "🔍 <span style='color:#28a745'>Đang kiểm tra trạng thái hệ thống...</span>";
        
        setTimeout(() => {
            msg.innerHTML = "✅ Hệ thống Online. Proxy sẵn sàng kết nối.";
        }, 1500);
    }
</script>
